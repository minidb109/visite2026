# VISITE 2026 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construir os 7 enigmas, o manual do Player 2 e o acabamento do jogo cooperativo VISITE 2026 a partir da base descrita no guia.

**Architecture:** Cubo de 4 faces com 7 painéis isolados; núcleo (`game.js` + `main.js`) já definido no guia detém estado/tempo/selos, cada enigma é um módulo JS+CSS que só escreve dentro do seu `.conteudo` e só resolve via `game.solve(id)`, manual é página estática sem JS.

**Tech Stack:** HTML + CSS + JavaScript vanilla com `type="module"`, sem build, servido via Live Server / `python -m http.server` / GitHub Pages.

**Spec:** `guia_desenvolvimento_visite_2026.md` (472 linhas, seções 1–9 — este plano argumenta a partir dele; executores leem ambos).

## Global Constraints

- Face tem 960 × 480 px; cada `.conteudo` útil tem ~440 × 380 px — enigma deve caber nisso.
- Só escrever dentro do próprio `#<id> .conteudo`; nunca mexer em outros painéis, topo ou cubo.
- Classes CSS com prefixo do enigma (`.a3-grade`, `.a3-botao`); variáveis de estado dentro do próprio módulo.
- Toda função de clique/tecla/envio começa com `if (game.isLocked("<id>")) return;` usando id em minúsculas `a1`…`a6`, `final`.
- Única forma de resolver: `game.solve("<id>")` uma vez; núcleo cuida de selo/classe.
- Reiniciar = recarregar a página (zera tudo); enigmas não precisam de código de reinício; não cancelar `setTimeout`.
- Tempo total 45 min (`45 * 1000 * 60`); vermelho nos 5 min finais; derrota mostra modal "Tempo esgotado".
- Não existe chat entre jogadores; manual nunca contém resposta direta.
- `type="module"` exige servidor HTTP (não duplo clique); `window.game` exposto só em dev e removido antes de entregar.
- Git: uma branch por tarefa (`feat/a3-grade`), PR curto, revisão do outro antes do merge na `main`.
- Textos/símbolos como constantes no topo do arquivo do enigma para conferência contra o manual.

```js
import { game } from '../core/game.js';

const SEQ_NORMAL = [2, 4, 8, 16];
const RESPOSTA = 1;
const MSG_ERRO = 'Essa resposta ainda não encaixa.';

const root = document.querySelector('#a1 .conteudo');

root.innerHTML = `
  <div class="a1-seq"></div>
  <div class="a1-linha">
    <input class="a1-campo" type="number" aria-label="Resposta do padrão">

function render() {
  const seq = inverted ? [...SEQ_NORMAL].reverse() : SEQ_NORMAL;
  root.querySelector('.a1-seq').textContent = seq.join(' → ') + ' → ?';
  root.querySelector('.a1-espelho').classList.toggle('ativo', inverted);
}
render();

root.querySelector('.a1-espelho').addEventListener('click', () => {
  if (game.isLocked('a1')) return;
  inverted = !inverted;
  render();
});

root.querySelector('.a1-ok').addEventListener('click', () => {
  if (game.isLocked('a1')) return;
  const raw = root.querySelector('.a1-campo').value;
  const n = Number(raw);
  if (inverted && raw !== '' && n === RESPOSTA) {
    game.solve('a1');
  } else {
    root.querySelector('.a1-msg').textContent = MSG_ERRO;
    shake(root);
  }
});
```

Regra de avaliação (Wordle): 1) dígito na posição certa → CERTO; 2) senão se existe em outra posição ainda não "usada" nesta tentativa → POSIÇÃO ERRADA; 3) senão AUSENTE. Repetição permitida.

- [ ] **Step 1: Escrever `js/puzzles/a2.js`**

```js
import { game } from '../core/game.js';
import { shake } from '../core/utils.js';

const CODIGO = ['0', '4', '2'];
const MAX_TENT = 5;
const SIMBOLOS = [
  { certo: '☀️', pos: '🌙', aus: '💧' },
  { certo: '⭐', pos: '🍃', aus: '⛰️' },
  { certo: '🔑', pos: '⚓', aus: '🔔' },
  { certo: '👑', pos: '👁️', aus: '⚡' },
  { certo: '🐚', pos: '🔥', aus: '☁️' },
];

const root = document.querySelector('#a2 .conteudo');
let tentativa = 1;

root.innerHTML = `
  <div class="a2-entradas">
    <input class="a2-d" maxlength="1" inputmode="numeric" aria-label="Dígito 1">
    <input class="a2-d" maxlength="1" inputmode="numeric" aria-label="Dígito 2">
    <input class="a2-d" maxlength="1" inputmode="numeric" aria-label="Dígito 3">
    <button class="a2-ok">Tentar</button>
  </div>
  <p class="a2-cont">Tentativa 1/5</p>
  <ul class="a2-lista"></ul>
  <p class="a2-msg" role="status"></p>`;

function avaliar(palpite) {
  const res = ['aus', 'aus', 'aus'];
  const usado = [false, false, false];
  for (let i = 0; i < 3; i++) {
    if (palpite[i] === CODIGO[i]) { res[i] = 'certo'; usado[i] = true; }
  }
  for (let i = 0; i < 3; i++) {
    if (res[i] === 'certo') continue;
    for (let j = 0; j < 3; j++) {
      if (!usado[j] && palpite[i] === CODIGO[j]) { res[i] = 'pos'; usado[j] = true; break; }
    }
  }
  return res;
}

root.querySelector('.a2-ok').addEventListener('click', () => {
  if (game.isLocked('a2')) return;
  const digs = [...root.querySelectorAll('.a2-d')].map(el => el.value.trim());
  if (digs.some(d => !/^[0-9]$/.test(d))) {
    root.querySelector('.a2-msg').textContent = 'Digite 3 dígitos de 0 a 9.';
    shake(root);
    return;
  }
  const linha = tentativa;
  const aval = avaliar(digs);
  if (digs.join('') === CODIGO.join('')) { game.solve('a2'); return; }
  const simb = SIMBOLOS[linha - 1];
  const li = document.createElement('li');
  li.textContent = `${digs.join('')} — ${aval.map(a => simb[a === 'certo' ? 'certo' : a === 'pos' ? 'pos' : 'aus']).join(' ')}`;
  root.querySelector('.a2-lista').appendChild(li);
  if (tentativa >= MAX_TENT) {
    root.querySelector('.a2-msg').textContent = 'O cadeado reinicia';
    root.querySelector('.a2-lista').innerHTML = '';
    tentativa = 1;
  } else {
    tentativa++;
    root.querySelector('.a2-msg').textContent = '';
  }
  root.querySelector('.a2-cont').textContent = `Tentativa ${tentativa}/5`;
});
```

```js
import { game } from '../core/game.js';

const SERIAL = 'KV-5291';
const ORDEM_CERTA = ['A4', 'D1', 'A1', 'D4'];
const SIMBOLOS = {
  A1: '★', A2: '🌙', A3: '☀️', A4: '🔑',
  B1: '🎵', B2: '⭕', B3: '⚡', B4: '🌸',
  C1: '✖️', C2: '🔺', C3: '🔷', C4: '🍃',
  D1: '✝️', D2: '🍀', D3: '❤️', D4: '⬛',
};
const ZONA = ['B2', 'B3', 'C2', 'C3'];

const root = document.querySelector('#a3 .conteudo');
let progresso = [];
let erros = 0;
let bloqueado = false;

root.innerHTML = `
  <p class="a3-serie">Série ${SERIAL} · erros <span class="a3-erros">0/3</span></p>
  <div class="a3-duplo">
    <div class="a3-grade" role="group" aria-label="Botões"></div>
    <div class="a3-leds" aria-label="Zona de conforto"></div>
  </div>
  <p class="a3-msg" role="status"></p>`;

const grade = root.querySelector('.a3-grade');
const leds = root.querySelector('.a3-leds');
for (const lin of ['A', 'B', 'C', 'D']) {
  for (let c = 1; c <= 4; c++) {
    const id = lin + c;
    const b = document.createElement('button');
    b.className = 'a3-botao';
    b.dataset.pos = id;
    b.textContent = SIMBOLOS[id];
    b.setAttribute('aria-label', id);
    grade.appendChild(b);
    const led = document.createElement('div');
    led.className = 'a3-led' + (ZONA.includes(id) ? ' aceso' : '');
    led.dataset.pos = id;
    leds.appendChild(led);
  }
}
function pintar() {
  root.querySelector('.a3-erros').textContent = `${erros}/3`;
  grade.querySelectorAll('.a3-botao').forEach(b => {
    b.classList.toggle('marcado', progresso.includes(b.dataset.pos));
  });
}
pintar();

grade.addEventListener('click', (ev) => {
  if (game.isLocked('a3')) return;
  if (bloqueado) return;
  const btn = ev.target.closest('.a3-botao');
  if (!btn) return;
  const pos = btn.dataset.pos;
  const esperado = ORDEM_CERTA[progresso.length];
  if (pos === esperado) {
    progresso.push(pos);
    pintar();
    if (progresso.length === ORDEM_CERTA.length) game.solve('a3');
  } else {
    erros++;
    progresso = [];
    if (erros >= 3) {
      erros = 0;
      bloqueado = true;
      root.querySelector('.a3-msg').textContent = 'Zona instável — aguarde 3 s';
      leds.classList.add('piscar');
      setTimeout(() => { bloqueado = false; leds.classList.remove('piscar'); root.querySelector('.a3-msg').textContent = ''; }, 3000);
    }
    pintar();
  }
});
```

```js
import { game } from '../core/game.js';

const INICIO = 'MORA';
const ALVO = 'RAMO';
const MAX_AP = 4;
const FUNCAO = {
  1: { 1: 'ESQ', 2: 'EXT', 3: 'PAR', 4: 'MEI' },
  2: { 1: 'MEI', 2: 'DIR', 3: 'EXT', 4: 'PAR' },
  3: { 1: 'MEI', 2: 'ESQ', 3: 'PAR', 4: 'EXT' },
  4: { 1: 'MEI', 2: 'DIR', 3: 'PAR', 4: 'EXT' },
};
const OPS = {
  ESQ: (w) => w.slice(1) + w[0],
  DIR: (w) => w[3] + w.slice(0, 3),
  PAR: (w) => w[1] + w[0] + w[3] + w[2],
  EXT: (w) => w[3] + w.slice(1, 3) + w[0],
  MEI: (w) => w[0] + w[2] + w[1] + w[3],
};

const root = document.querySelector('#a4 .conteudo');
let palavra = INICIO;
let n = 0;

root.innerHTML = `
  <p class="a4-palavra">${palavra}</p>
  <p class="a4-alvo">Alvo: ${ALVO}</p>
  <div class="a4-botoes">
    <button data-b="1">1</button><button data-b="2">2</button>
    <button data-b="3">3</button><button data-b="4">4</button>
  </div>
  <p class="a4-cont">0/4</p>
  <p class="a4-msg" role="status"></p>`;

function alcancavel(pal, usados) {
  if (pal === ALVO) return true;
  if (usados >= MAX_AP) return false;
  const prox = usados + 1;
  for (let b = 1; b <= 4; b++) {
    if (alcancavel(OPS[FUNCAO[prox][b]](pal), prox)) return true;
  }
  return false;
}
function reset(msg) {
  palavra = INICIO; n = 0;
  root.querySelector('.a4-palavra').textContent = palavra;
  root.querySelector('.a4-cont').textContent = '0/4';
  root.querySelector('.a4-msg').textContent = msg;
}
root.querySelector('.a4-botoes').addEventListener('click', (ev) => {
  if (game.isLocked('a4')) return;
  const btn = ev.target.closest('button');
  if (!btn) return;
  const fn = FUNCAO[n + 1][btn.dataset.b];
  palavra = OPS[fn](palavra);
  n++;
  root.querySelector('.a4-palavra').textContent = palavra;
  root.querySelector('.a4-cont').textContent = `${n}/4`;
  if (palavra === ALVO) { game.solve('a4'); return; }
  if (n >= MAX_AP || !alcancavel(palavra, n)) reset('A palavra retorna ao início.');
});
```
